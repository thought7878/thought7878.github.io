```bash
#!/bin/bash
# remove_audio_subs.sh — 移除视频的音频流与软字幕流，保留画面（无损流复制）
# 用法: ./remove_audio_subs.sh <视频文件|目录>

set -u

SUFFIX="_无音轨无软字幕"
VIDEO_EXTS="mp4 mkv mov avi m4v webm flv ts wmv mpg mpeg 3gp ogv"
GENERATED=()

err()  { echo "[错误] $*" >&2; }
info() { echo "[信息] $*"; }

usage_exit() {
  echo "用法: $0 <视频文件或目录>"
  exit 1
}

# ---------- 参数与环境校验 ----------
[ $# -eq 1 ] || { err "必须且只能提供一个参数（视频文件或目录）"; usage_exit; }

INPUT="$1"

if ! command -v ffmpeg >/dev/null 2>&1; then
  err "未找到 ffmpeg，请先安装（例如: brew install ffmpeg）"
  exit 1
fi

if [ ! -e "$INPUT" ]; then
  err "输入路径不存在: $INPUT"
  exit 1
fi

is_video_file() {
  local f="$1" ext
  ext="${f##*.}"
  ext=$(echo "$ext" | tr '[:upper:]' '[:lower:]')
  for e in $VIDEO_EXTS; do
    [ "$ext" = "$e" ] && return 0
  done
  return 1
}

# ---------- 单文件处理 ----------
process_file() {
  local src="$1"
  local dir base name ext out

  dir=$(dirname "$src")
  base=$(basename "$src")
  ext="${base##*.}"
  name="${base%.*}"
  out="$dir/${name}${SUFFIX}.${ext}"

  if [ "$out" = "$src" ]; then
    err "输出与源文件同名，跳过: $src"
    return 1
  fi

  if [ -e "$out" ]; then
    info "输出已存在，跳过（不覆盖）: $out"
    return 2
  fi

  # -map 0:V 仅保留真实视频流（排除封面图等附加图片流）
  # -c:v copy 无损复制视频，画质不变；-an -sn -dn 去除音频/软字幕/数据流
  if ffmpeg -hide_banner -loglevel error -i "$src" \
       -map 0:V -c:v copy -an -sn -dn \
       -- "$out" 2>/tmp/ffmpeg_err_$$.log; then
    info "完成: $out"
    GENERATED+=("$out")
    return 0
  else
    err "处理失败: $src"
    sed 's/^/       /' /tmp/ffmpeg_err_$$.log >&2
    rm -f "$out" /tmp/ffmpeg_err_$$.log
    return 1
  fi
}

# ---------- 收集目标文件 ----------
FILES=()
if [ -d "$INPUT" ]; then
  while IFS= read -r -d '' f; do
    FILES+=("$f")
  done < <(find "$INPUT" -type f -print0)
else
  FILES+=("$INPUT")
fi

if [ ${#FILES[@]} -eq 0 ]; then
  err "目录下没有找到任何文件: $INPUT"
  exit 1
fi

# ---------- 批量执行 ----------
ok=0; skip=0; fail=0
for f in "${FILES[@]}"; do
  if [ -d "$INPUT" ] && ! is_video_file "$f"; then
    continue   # 目录模式：跳过非视频文件
  fi
  # 跳过本脚本已生成过的文件（避免套娃处理）
  case "$(basename "$f")" in
    *"$SUFFIX"*) info "已是处理后文件，跳过: $f"; continue ;;
  esac
  if process_file "$f"; then
    ok=$((ok+1))
  else
    rc=$?
    if [ $rc -eq 2 ]; then skip=$((skip+1)); else fail=$((fail+1)); fi
  fi
done

# ---------- 结果汇总 ----------
echo
echo "========== 处理结果 =========="
echo "成功: $ok  跳过: $skip  失败: $fail"
if [ ${#GENERATED[@]} -gt 0 ]; then
  echo "生成的文件："
  for g in "${GENERATED[@]}"; do
    echo "  $g"
  done
fi

[ $fail -eq 0 ] || exit 1
exit 0
```
